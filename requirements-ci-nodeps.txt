# Пакеты, которые CI ставит БЕЗ их зависимостей: `pip install --no-deps -r`.
#
# Зачем отдельный файл: `--no-deps` — флаг всей команды, а не отдельной строки в
# requirements-файле. Поставить один пакет без зависимостей, а остальные с ними,
# можно только вторым вызовом pip, и список для этого вызова лежит здесь.
#
# Правило, по которому пакет попадает сюда: из него нужен только бинарник,
# питонья часть не импортируется ни кодом, ни тестами. Тогда её зависимости —
# чистый вес на трёх раннерах и лишний риск: смежный пакет тянет свою версию
# numpy и ломает пин, из-за которого сломается уже не установка, а тесты.
#
# Версии, как и в requirements-ci.txt, обязаны совпадать с pyproject.toml —
# за этим следит tests/unit/test_packaging.py.

# pyrnnoise кладёт рядом со своим кодом собранную RNNoise (rnnoise.dll под
# windows, librnnoise.so под linux и macOS), и ayris.audio.denoise грузит её
# через ctypes по пути внутри пакета. Сам пакет при этом не импортируется, а его
# Requires-Dist — audiolab, click, matplotlib, numpy, tqdm — нужны только его
# собственной обёртке и утилите командной строки.
pyrnnoise==0.4.3
